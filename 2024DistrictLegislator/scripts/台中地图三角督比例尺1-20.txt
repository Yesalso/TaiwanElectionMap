# ===================== 工具函数 =====================
def strip_town_suffix(s):
    return "" if pd.isna(s) else re.sub(r"[鄉鎮市區]$", "", str(s).strip())

def strip_village_suffix(s):
    return "" if pd.isna(s) else re.sub(r"[村里]$", "", str(s).strip())

def get_color_by_value(val, stops):
    if np.isnan(val): return None
    for upper, hx in stops:
        if val <= upper: return hx
    return stops[-1][1]

def hex2rgb(hx):
    h = hx.lstrip("#")
    return (int(h[0:2],16)/255.0, int(h[2:4],16)/255.0, int(h[4:6],16)/255.0)

def text_width(font, txt):
    try:
        bbox = font.getbbox(txt)
    except AttributeError:
        bbox = font.getmask(txt).getbbox()
    return (bbox[2]-bbox[0]) if bbox else 0

# ===================== 填色 =====================
def pick_fill_color(row):
    r1, r2, r3 = row["rate1"], row["rate2"], row["rate3"]
    if np.isnan(r1) and np.isnan(r2) and np.isnan(r3):
        return GRAY_COLOR
    if np.isnan(r1) or np.isnan(r2) or np.isnan(r3):
        return np.nan
    if r1 >= r2 and r1 >= r3:
        return get_color_by_value(r1, color_stops_rate1)
    elif r2 >= r3:
        return get_color_by_value(r2, color_stops_rate2)
    else:
        return get_color_by_value(r3, color_stops_rate3)

gdf_nt["fill_hex"] = gdf_nt.apply(pick_fill_color, axis=1)
gdf_plot = gdf_nt[~gdf_nt["fill_hex"].isna()].copy()

    raise ValueError("没有可绘制的村里，请检查数据匹配！")

# ===================== 颜色量化 =====================
allowed_hex = {"#FFFFFF", "#000000", GRAY_COLOR}
for stops in (color_stops_rate1, color_stops_rate2, color_stops_rate3):
    allowed_hex.update(hx for _, hx in stops)
allowed_rgb_255 = np.array([hex2rgb(hx) for hx in allowed_hex]).astype(np.float32) * 255

pixels = img_rgb.reshape(-1, 3).astype(np.float32)
distances = np.linalg.norm(pixels[:, None, :] - allowed_rgb_255[None, :, :], axis=2)
quantized = allowed_rgb_255[np.argmin(distances, axis=1)].astype(np.uint8).reshape(img_rgb.shape)

# ===================== 图例 =====================
pil_img = Image.fromarray(quantized)
W, H = pil_img.size

label_font = None
for path in ["C:/Windows/Fonts/msyh.ttc", "C:/Windows/Fonts/simhei.ttf", "C:/Windows/Fonts/mingliu.ttc"]:
    try:
        label_font = ImageFont.truetype(path, LABEL_FONT_SIZE)
        break
    except Exception:
        continue
if label_font is None:
    label_font = ImageFont.load_default()

def get_label_text(upper, prev=None):
    return f"≤{upper}%" if prev is None else f"{prev}~{upper}%"

stops_list = [color_stops_rate1, color_stops_rate2, color_stops_rate3]
max_label_w = max(
    text_width(label_font, get_label_text(u, stops[i-1][0] if i > 0 else None))
    for stops in stops_list for i, (u, _) in enumerate(stops)
)
max_label_w = max(max_label_w, text_width(label_font, "无匹配数据"))

col_width = BLOCK_WIDTH + 5 + max_label_w
total_legend_width = col_width * 3 + H_SPACING * 2
max_n = max(len(s) for s in stops_list)
legend_height = (max_n + 1) * (BLOCK_HEIGHT + V_SPACING) - V_SPACING

new_W = W + total_legend_width + LEGEND_PADDING * 2
new_H = max(H, legend_height + LEGEND_PADDING * 2)

new_img = Image.new('RGB', (new_W, new_H), (255, 255, 255))
new_img.paste(pil_img, (LEGEND_PADDING, (new_H - H) // 2))

legend_x = W + LEGEND_PADDING
legend_y = (new_H - legend_height) // 2
col_positions = [legend_x + i * (col_width + H_SPACING) for i in range(3)]
draw = ImageDraw.Draw(new_img)

def draw_swatch(x, y, color_hex, label):
    draw.rectangle([x, y, x + BLOCK_WIDTH, y + BLOCK_HEIGHT],
                   fill=color_hex, outline="#000000", width=1)
    try:
        bbox = label_font.getbbox(label)
    except AttributeError:
        bbox = label_font.getmask(label).getbbox()
    label_y = y + (BLOCK_HEIGHT - (bbox[3] - bbox[1])) // 2 if bbox else y
    draw.text((x + BLOCK_WIDTH + 5, label_y), label, fill="#000000", font=label_font)

for col_idx, stops in enumerate(stops_list):
    x_start = col_positions[col_idx]
    for i, (upper, color_hex) in enumerate(stops):
        y = legend_y + i * (BLOCK_HEIGHT + V_SPACING)
        prev = stops[i-1][0] if i > 0 else None
        draw_swatch(x_start, y, color_hex, get_label_text(upper, prev))

draw_swatch(col_positions[0],
            legend_y + max_n * (BLOCK_HEIGHT + V_SPACING),
            GRAY_COLOR, "无匹配数据")

new_img.save(OUT_PNG)

print(f"   边线: 村里1px | 选区白线×{DISTRICT_LINE_WIDTH_FACTOR} | 外轮廓黑线")
print(f"   着色: 三率取大值对应色阶 | 无匹配→{GRAY_COLOR}")
