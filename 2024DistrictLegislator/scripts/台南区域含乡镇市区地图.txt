VARIANT_MAP = str.maketrans({'濓':'濂', '𥕢':'曹', '𦰡':'那', '檨':'檨'})
_CJK_KEEP = re.compile(
    r'[^\u3400-\u4DBF\u4E00-\u9FFF\uF900-\uFAFF'
    r'\U00020000-\U0002A6DF\U0002A700-\U0002B73F'

def normalize_text(s):
    if pd.isna(s): return ""
    s = str(s).strip()
    s = unicodedata.normalize('NFKC', s)
    s = ''.join(c for c in s if unicodedata.category(c)[0] != 'M')
    s = re.sub(r'[\uFE00-\uFE0F\U000E0100-\U000E01EF]', '', s)
    s = re.sub(r'[\u200B-\u200F\u202A-\u202E\u2060-\u206F\uFEFF]', '', s)
    return s.translate(VARIANT_MAP)

def strip_town(s):  return "" if pd.isna(s) else re.sub(r"[鄉鎮市區]$", "", normalize_text(s))
def strip_vill(s):  return "" if pd.isna(s) else re.sub(r"[村里]$", "", normalize_text(s))
def clean_cjk(s):   return "" if pd.isna(s) else _CJK_KEEP.sub('', normalize_text(s))

def hex2rgb(hx):
    h = hx.lstrip("#")
    return (int(h[0:2],16)/255.0, int(h[2:4],16)/255.0, int(h[4:6],16)/255.0)

def get_color(val, stops):
    if np.isnan(val): return None
    for up, hx in stops:
        if val <= up: return hx
    return stops[-1][1]

def text_width(font, txt):
    try: bbox = font.getbbox(txt)
    except AttributeError: bbox = font.getmask(txt).getbbox()
    return (bbox[2]-bbox[0]) if bbox else 0

def fill_small_white_blobs(img, protect_mask=None):
    h, w = img.shape[:2]
    mask = np.all(img >= WHITE_THRESH, axis=2).astype(np.uint8)
    num, labels, stats, _ = cv2.connectedComponentsWithStats(mask, connectivity=8)
    res = img.copy(); filled = 0
    kernel = np.ones((3, 3), np.uint8)
    for i in range(1, num):
        x, y, bw, bh, area = stats[i]
        if area > MAX_WHITE_BLOB_PX: continue
        if x == 0 or y == 0 or (x+bw) == w or (y+bh) == h: continue
        cm = (labels == i).astype(np.uint8)
        if protect_mask is not None and np.any(protect_mask & cm.astype(bool)): continue
        dil = cv2.dilate(cm, kernel, iterations=RING_ITER)
        rm = (dil > 0) & (cm == 0)
        rp = img[rm]
        if len(rp) == 0: continue
        nw = rp[~np.all(rp >= WHITE_THRESH, axis=1)]
        if len(nw) == 0: continue
        colors, counts = np.unique(nw.reshape(-1, 3), axis=0, return_counts=True)
        res[cm.astype(bool)] = colors[np.argmax(counts)]
        filled += 1
    return res, filled

MIN_SIM = 0.5

def fuzzy_lookup(town, vill, by_key, by_town):
    if not town or not vill: return None, None, "none"
    tn, vn = normalize_text(town), normalize_text(vill)
    if (tn, vn) in by_key:
        return vill, by_key[(tn, vn)], "exact"
    vc = clean_cjk(vn)
    if not vc: return None, None, "none"
    best_v, best_val, best_r = None, None, -1.0
    for cand, val in by_town.get(tn, []):
        cc = clean_cjk(cand)
        if not cc: continue
        if cc == vc: return cand, val, f"fuzzy:{cand}(1.00)"
        r = SequenceMatcher(None, vc, cc).ratio()
        if r > best_r: best_r, best_v, best_val = r, cand, val
    if best_val is None or best_r < MIN_SIM: return None, None, "none"
    return best_v, best_val, f"fuzzy:{best_v}({best_r:.2f})"

    district_by_town.setdefault(t, []).append((v, d))

def process_row(r):
    town, vill = r["town_core"], r["vill_core"]
    mv, vals, mt = fuzzy_lookup(town, vill, vote_dict, vote_by_town)
    r1, r2, r3 = vals if vals else (np.nan, np.nan, np.nan)
    lv = mv if mv is not None else vill
    d = village2district.get((town, lv))
    if d is None: d = district_full_town.get(town)
    if d is None:
        _, dv, _ = fuzzy_lookup(town, lv, village2district, district_by_town)
        d = dv if dv is not None else np.nan
    return (r1, r2, r3, mt, d)
