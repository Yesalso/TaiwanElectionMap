def strip_town_suffix(s):
    if pd.isna(s):
        return ""
    s = str(s).strip()
    s = re.sub(r"[鄉鎮市區]$", "", s)
    return s

def strip_village_suffix(s):
    if pd.isna(s):
        return ""
    s = str(s).strip()
    s = re.sub(r"[村里]$", "", s)
    return s

def get_color_by_value(val, stop_list):
    if np.isnan(val):
        return None
    for upper, hx in stop_list:
        if val <= upper:
            return hx
    return stop_list[-1][1]

def hex2rgb(hx):
    h = hx.lstrip("#")
    return (int(h[0:2],16)/255.0, int(h[2:4],16)/255.0, int(h[4:6],16)/255.0)

vote_dict = {}
for _, row in df_vote.iterrows():
    k = (row["town_core"], row["vill_core"])

def get_three_rates(shp_row):

    if key in vote_dict:
        return pd.Series(vote_dict[key])
    else:
        return pd.Series((np.nan, np.nan, np.nan))

# 填色逻辑
def pick_fill_color(row):
    r1, r2, r3 = row["rate1"], row["rate2"], row["rate3"]
    if np.isnan(r1) or np.isnan(r2) or np.isnan(r3):
        return np.nan
    if r1 >= r2 and r1 >= r3:
        return get_color_by_value(r1, color_stops_rate1)
    elif r2 >= r1 and r2 >= r3:
        return get_color_by_value(r2, color_stops_rate2)
    else:
        return get_color_by_value(r3, color_stops_rate3)

gdf_nt["fill_hex"] = gdf_nt.apply(pick_fill_color, axis=1)

gdf_plot = gdf_nt[~gdf_nt["fill_hex"].isna()].copy()

    raise ValueError("没有可绘制的村里，请检查数据匹配！")

fill_colors = gdf_plot["fill_hex"].tolist()
minx, miny, maxx, maxy = gdf_plot.total_bounds

allowed_hex = set()
allowed_hex.add("#FFFFFF")

    allowed_hex.add(hx)

    allowed_hex.add(hx)
for _, hx in color_stops_rate3:
    allowed_hex.add(hx)

allowed_rgb = np.array([hex2rgb(hx) for hx in allowed_hex])
allowed_rgb_255 = (allowed_rgb * 255).astype(np.uint8)
pixels = img_rgb.reshape(-1, 3).astype(np.float32)
distances = np.linalg.norm(pixels[:, None, :] - allowed_rgb_255[None, :, :], axis=2)
nearest_idx = np.argmin(distances, axis=1)
quantized = allowed_rgb_255[nearest_idx].reshape(img_rgb.shape)

# ========== PIL图例（三列） ==========
pil_img = Image.fromarray(quantized)
W, H = pil_img.size

font_paths = [
    "C:/Windows/Fonts/msyh.ttc",
    "C:/Windows/Fonts/simhei.ttf",
    "C:/Windows/Fonts/mingliu.ttc",
]
label_font = None
for path in font_paths:
    try:
        label_font = ImageFont.truetype(path, LABEL_FONT_SIZE)
        break
    except:
        continue
if label_font is None:
    label_font = ImageFont.load_default()

def get_label_text(upper, prev_upper=None):
    if prev_upper is None:
        return f"≤{upper}%"
    else:
        return f"{prev_upper}~{upper}%"

stops_list = [color_stops_rate1, color_stops_rate2, color_stops_rate3]

max_label_w = 0
for stops in stops_list:
    for i, (upper, _) in enumerate(stops):
        prev = stops[i-1][0] if i > 0 else None
        label = get_label_text(upper, prev)
        try:
            bbox = label_font.getbbox(label)
        except AttributeError:
            bbox = label_font.getmask(label).getbbox()
        if bbox:
            w = bbox[2] - bbox[0]
            max_label_w = max(max_label_w, w)

col_width = BLOCK_WIDTH + 5 + max_label_w
total_legend_width = col_width * 3 + H_SPACING * 2
max_n = max(len(stops) for stops in stops_list)
legend_height = max_n * (BLOCK_HEIGHT + V_SPACING) - V_SPACING

new_W = W + total_legend_width + LEGEND_PADDING * 2
new_H = max(H, legend_height + LEGEND_PADDING * 2)

new_img = Image.new('RGB', (new_W, new_H), (255, 255, 255))
y_offset = (new_H - H) // 2
new_img.paste(pil_img, (LEGEND_PADDING, y_offset))

legend_x = W + LEGEND_PADDING
legend_y = (new_H - legend_height) // 2

col_positions = [legend_x + i * (col_width + H_SPACING) for i in range(3)]
draw = ImageDraw.Draw(new_img)

for col_idx, stops in enumerate(stops_list):
    x_start = col_positions[col_idx]
    for i, (upper, color_hex) in enumerate(stops):
        y = legend_y + i * (BLOCK_HEIGHT + V_SPACING)
        draw.rectangle([x_start, y, x_start + BLOCK_WIDTH, y + BLOCK_HEIGHT],
                       fill=color_hex, outline="#000000", width=1)
        prev = stops[i-1][0] if i > 0 else None
        label = get_label_text(upper, prev)
        label_x = x_start + BLOCK_WIDTH + 5
        try:
            bbox = label_font.getbbox(label)
        except AttributeError:
            bbox = label_font.getmask(label).getbbox()
        if bbox:
            label_h = bbox[3] - bbox[1]
            label_y = y + (BLOCK_HEIGHT - label_h) // 2
        else:
            label_y = y
        draw.text((label_x, label_y), label, fill="#000000", font=label_font)

new_img.save(OUT_PNG)
print(f"✅完成输出：{OUT_PNG}  新尺寸 {new_W}×{new_H} px")
print(f"👉边线1像素，颜色量化，图例位于右侧空白区域")
print(f"👉色块尺寸：宽{BLOCK_WIDTH}px 高{BLOCK_HEIGHT}px，列间距{H_SPACING}px，行间距{V_SPACING}px")
print(f"👉填充规则：取三个得票率中的最大值对应的色阶进行着色")
