import unicodedata

VARIANT_CHAR_MAP = str.maketrans({
    '濓': '濂',
    '𥕢': '曹',
})

def normalize_text(s):
    if pd.isna(s):
        return ""
    s = str(s).strip()
    s = unicodedata.normalize('NFKC', s)
    s = ''.join(c for c in s if unicodedata.category(c)[0] != 'M')
    s = re.sub(r'[\uFE00-\uFE0F\U000E0100-\U000E01EF]', '', s)
    s = re.sub(r'[\u200B-\u200F\u202A-\u202E\u2060-\u206F\uFEFF]', '', s)
    s = s.translate(VARIANT_CHAR_MAP)
    return s

def strip_town_suffix(s):
    return "" if pd.isna(s) else re.sub(r"[鄉鎮市區]$", "", normalize_text(s))

def strip_village_suffix(s):
    return "" if pd.isna(s) else re.sub(r"[村里]$", "", normalize_text(s))

def get_color_by_value(val, stops):
    if np.isnan(val):
        return None
    for upper, hx in stops:
        if val <= upper:
            return hx
    return stops[-1][1]

def hex2rgb(hx):
    h = hx.lstrip("#")
    return (int(h[0:2],16)/255.0, int(h[2:4],16)/255.0, int(h[4:6],16)/255.0)

def text_width(font, txt):
    try:
        bbox = font.getbbox(txt)
    except AttributeError:
        bbox = font.getmask(txt).getbbox()
    return (bbox[2]-bbox[0]) if bbox else 0

# ===================== 精确 + 同乡镇最相似模糊匹配 =====================
MIN_SIMILARITY = 0.5

def fuzzy_lookup(town, vill, by_key, by_town):
    if not town or not vill:
        return None, None, "none"
    key = (town, vill)
    if key in by_key:
        return vill, by_key[key], "exact"
    best_v, best_val, best_ratio = None, None, -1.0
    for cand_v, val in by_town.get(town, []):
        ratio = SequenceMatcher(None, vill, cand_v).ratio()
        if ratio > best_ratio:
            best_ratio, best_v, best_val = ratio, cand_v, val
    if best_val is None or best_ratio < MIN_SIMILARITY:
        return None, None, "none"
    return best_v, best_val, f"fuzzy:{best_v}({best_ratio:.2f})"

    return pd.Series([r1, r2, r3, excel_mt])

# 将 RGB 编码成 int32，压缩为唯一色表，再算唯一色 × 允许色 的距离矩阵
pixels_flat = img_rgb.reshape(-1, 3).astype(np.uint8)
codes = (pixels_flat[:, 0].astype(np.int32) << 16) | \
        (pixels_flat[:, 1].astype(np.int32) << 8)  | \
         pixels_flat[:, 2].astype(np.int32)

unique_codes, inverse = np.unique(codes, return_inverse=True)
unique_colors = np.stack(
    [(unique_codes >> 16) & 0xFF,
     (unique_codes >> 8)  & 0xFF,
      unique_codes        & 0xFF],
    axis=1
).astype(np.float32)                       # (U, 3)，U 通常只有几千

uniq_dist    = np.linalg.norm(unique_colors[:, None, :] - allowed_rgb_255[None, :, :], axis=2)
uniq_nearest = allowed_rgb_255[np.argmin(uniq_dist, axis=1)]   # (U, 3)

def draw_legend_on(draw_obj, x0, y0, font):
    col_positions = [x0 + i * (col_width + H_SPACING) for i in range(3)]
    for col_idx, stops in enumerate(stops_list):
        x_start = col_positions[col_idx]
        for i, (upper, color_hex) in enumerate(stops):
            y = y0 + i * (BLOCK_HEIGHT + V_SPACING)
            draw_obj.rectangle(
                [x_start, y, x_start + BLOCK_WIDTH, y + BLOCK_HEIGHT],
                fill=color_hex, outline="#000000", width=1)
            prev = stops[i - 1][0] if i > 0 else None
            label = get_label_text(upper, prev)
            label_x = x_start + BLOCK_WIDTH + 5
            try:
                bbox = font.getbbox(label)
            except AttributeError:
                bbox = font.getmask(label).getbbox()
            label_y = y + (BLOCK_HEIGHT - (bbox[3] - bbox[1])) // 2 if bbox else y
            draw_obj.text((label_x, label_y), label, fill="#000000", font=font)

    # 无匹配数据：只在第一列下方画一次
    y = y0 + max_n * (BLOCK_HEIGHT + V_SPACING)
    x_start = col_positions[0]
    draw_obj.rectangle(
        [x_start, y, x_start + BLOCK_WIDTH, y + BLOCK_HEIGHT],
        fill=GRAY_COLOR, outline="#000000", width=1)
    try:
        bbox = font.getbbox("无匹配数据")
    except AttributeError:
        bbox = font.getmask("无匹配数据").getbbox()
    label_y = y + (BLOCK_HEIGHT - (bbox[3] - bbox[1])) // 2 if bbox else y
    draw_obj.text((x_start + BLOCK_WIDTH + 5, label_y),
                  "无匹配数据", fill="#000000", font=font)
