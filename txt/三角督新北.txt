def strip_town_suffix(s):
    """去掉鄉鎮市區结尾，处理板橋市/板橋區改制差异"""
    if pd.isna(s):
        return ""
    s = str(s).strip()
    s = re.sub(r"[鄉鎮市區]$", "", s)
    return s

def strip_village_suffix(s):
    """村里剥离结尾：里、村，处理村里改制村改里"""
    if pd.isna(s):
        return ""
    s = str(s).strip()
    s = re.sub(r"[村里]$", "", s)
    return s

# 根据数值取对应色阶的hex颜色
def get_color_by_value(val, stop_list):
    if np.isnan(val):
        return None
    for upper, hx in stop_list:
        if val <= upper:
            return hx
    return stop_list[-1][1]   # 超过最大值使用最后一个颜色

def hex2rgb(hx):
    h = hx.lstrip("#")
    return (int(h[0:2],16)/255.0, int(h[2:4],16)/255.0, int(h[4:6],16)/255.0)

# 赋值三个率
def get_three_rates(shp_row):
    t_core = shp_row["town_core"]
    v_core = shp_row["vill_core"]
    key = (t_core, v_core)
    if key in vote_dict:
        return pd.Series(vote_dict[key])
    else:
        return pd.Series((np.nan, np.nan, np.nan))

# 填色逻辑：rate1 >= rate2 用 rate1 色阶，否则用 rate2 色阶
# 任一为空则填白色（#FFFFFF）

def pick_fill_color(row):
    r1 = row["rate1"]
    r2 = row["rate2"]
    if np.isnan(r1) or np.isnan(r2):
        return np.nan
    if r1 >= r2:
        return get_color_by_value(r1, color_stops_rate1)
    else:
        return get_color_by_value(r2, color_stops_rate2)

gdf_nt["fill_hex"] = gdf_nt.apply(pick_fill_color, axis=1)
