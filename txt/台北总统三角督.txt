def strip_town_suffix(s):
    if pd.isna(s):
        return ""
    s = str(s).strip()
    s = re.sub(r"[鄉鎮市區]$", "", s)
    return s

def strip_village_suffix(s):
    if pd.isna(s):
        return ""
    s = str(s).strip()
    s = re.sub(r"[村里]$", "", s)
    return s

def get_color_by_value(val, stop_list):
    if np.isnan(val):
        return None
    for upper, hx in stop_list:
        if val <= upper:
            return hx
    return stop_list[-1][1]

def hex2rgb(hx):
    h = hx.lstrip("#")
    return (int(h[0:2],16)/255.0, int(h[2:4],16)/255.0, int(h[4:6],16)/255.0)

# ---------- 空白区域检测（严格版） ----------
def find_blank_region(rgb_array, rect_w, rect_h, margin):
    h, w = rgb_array.shape[:2]
    if rect_w + 2*margin > w or rect_h + 2*margin > h:
        return None

    non_white = (~np.all(rgb_array >= 250, axis=2)).astype(np.uint8)

    max_dim = 1000
    step = max(1, int(np.ceil(max(w, h) / max_dim)))
    if step > 1:
        pad_h = (-h) % step
        pad_w = (-w) % step
        padded = np.pad(non_white, ((0, pad_h), (0, pad_w)),
                        mode="constant", constant_values=0)
        sh, sw = padded.shape[0] // step, padded.shape[1] // step
        small = padded.reshape(sh, step, sw, step).max(axis=(1, 3))
    else:
        small = non_white
        sh, sw = small.shape

    small_margin = max(1, int(np.ceil(margin / step)))
    ksize = 2 * small_margin + 1
    kernel = np.ones((ksize, ksize), np.uint8)
    dilated = cv2.dilate(small, kernel)

    rw = int(np.ceil(rect_w / step))
    rh = int(np.ceil(rect_h / step))
    if rw > sw or rh > sh:
        return None

    integral = cv2.integral(dilated)
    xs = np.arange(0, sw - rw + 1)
    if len(xs) == 0:
        return None
    for y in range(0, sh - rh + 1):
        total = (integral[y + rh, xs + rw] - integral[y, xs + rw]
                 - integral[y + rh, xs] + integral[y, xs])
        hit = np.where(total == 0)[0]
        if len(hit) > 0:
            x_small = int(xs[hit[0]])
            x0 = max(0, x_small * step - small_margin * step)
            y0 = max(0, y * step - small_margin * step)
            if x0 + rect_w > w or y0 + rect_h > h:
                continue
            return (x0, y0)
    return None


def verify_blank_region(rgb_array, x0, y0, rect_w, rect_h):
    h, w = rgb_array.shape[:2]
    if x0 < 0 or y0 < 0 or x0 + rect_w > w or y0 + rect_h > h:
        return False
    sub = rgb_array[y0:y0+rect_h, x0:x0+rect_w]
    return bool(np.all(sub >= 250))

def draw_legend_on(draw_obj, x0, y0, font):
    col_positions = [x0 + i * (col_width + H_SPACING) for i in range(3)]
    for col_idx, stops in enumerate(stops_list):
        x_start = col_positions[col_idx]
        for i, (upper, color_hex) in enumerate(stops):
            y = y0 + i * (BLOCK_HEIGHT + V_SPACING)
            draw_obj.rectangle(
                [x_start, y, x_start + BLOCK_WIDTH, y + BLOCK_HEIGHT],
                fill=color_hex, outline="#000000", width=1)
            prev = stops[i - 1][0] if i > 0 else None
            label = get_label_text(upper, prev)
            label_x = x_start + BLOCK_WIDTH + 5
            try:
                bbox = font.getbbox(label)
            except AttributeError:
                bbox = font.getmask(label).getbbox()
            if bbox:
                label_h = bbox[3] - bbox[1]
                label_y = y + (BLOCK_HEIGHT - label_h) // 2
            else:
                label_y = y
            draw_obj.text((label_x, label_y), label, fill="#000000", font=font)
